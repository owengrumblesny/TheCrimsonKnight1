function selectSlot(index: number) {

  if (!slot) return;
  if (!slot.itemId || !inventory.has(slot.itemId)) { notify('Slot unavailable'); return; }
  inventory.setSelectedSlot(index);
  equipment.setSelectedItem(slot.itemId);
  renderHotbarFromState();
}
function onKeyDown(event: KeyboardEvent) {
  if (ui.isTyping() || ui.modalBlocksGameplay()) return;
  if (event.repeat && ui.isOneShotBinding(event.code)) return;
  const index = bindings.hotbarKeyToIndex[event.code];
  if (index !== undefined) { selectSlot(index); return; }
  // Other bindings dispatch through existing game input manager.
}
